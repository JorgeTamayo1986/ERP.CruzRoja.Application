

create database InventarioCRCDB

go

use InventarioCRCDB

go

create table Rol(
idRol int primary key identity(1,1),
descripcion varchar(50),
esActivo bit,
fechaRegistro datetime default getdate()
)
go

create table TipoSalida(
idTipo int primary key identity(1,1),
Descripcion varchar(50),
esActivo bit
)

go

create table Usuario(
idUsuario int primary key identity(1,1),
nombreApellidos varchar(100),
correo varchar(40),
idRol int references Rol(idRol),
clave varchar(40),
esActivo bit
)

go

create table Categoria(
idCategoria int primary key identity(1,1),
descripcion varchar(50),
esActivo bit,
fechaRegistro datetime default getdate()
)


go 


CREATE TABLE Producto (
    id INT primary key identity(1,1),
    nombre VARCHAR(100),
    idCategoria int references Categoria(idCategoria),
    esActivo bit
);

go

CREATE TABLE DetalleProducto (
    Id INT primary key identity(1,1),
    ProductoId INT FOREIGN KEY REFERENCES Producto(Id),
    FechaVencimiento DATE,
    Stock INT,
    FechaIngreso DATETIME DEFAULT GETDATE()
);

create table NumeroDocumento(
    idNumeroDocumento int primary key identity(1,1),
    ultimo_Numero int not null,
    fechaRegistro datetime default getdate()
)
go

create table Salida(
    idSalida int primary key identity(1,1),
    numeroDocumento varchar(40),
    fechaRegistro datetime default getdate()
)
go

create table DetalleSalida(
    idDetalleVenta int primary key identity(1,1),
    idVenta int references Salida(idSalida),
    idProducto int references Producto(id),
    tipoSalida Int FOREIGN KEY REFERENCES TipoSalida(idTipo),
    cantidad int
)

go
